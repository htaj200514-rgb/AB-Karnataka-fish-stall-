Here is the complete full-stack web application for **A B Karnataka Fish Stall**. The codebase includes a Node.js + Express backend with JSON Web Token (JWT) authentication, static asset serving, RESTful endpoints, and a front-end HTML/CSS/JavaScript interface.
### File Structure
```text

├── package.json
├── server.js
├── .env
└── public/
    ├── index.html
    ├── style.css
    └── app.js

```

// Authentication Middleware
const authenticateOwner = (req, res, next) => {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];
  
  if (!token) {
    return res.status(401).json({ success: false, message: "Unauthorized: Access denied." });
  }

  jwt.verify(token, JWT_SECRET, (err, user) => {
    if (err || user.email !== OWNER_EMAIL || user.role !== 'admin') {
      return res.status(403).json({ success: false, message: "Unauthorized: Only registered store owners can access this portal." });
    }
    req.user = user;
    next();
  });
};

// Authentication Routes
app.post('/api/auth/login', (req, res) => {
  const { email, password, role } = req.body;

  if (role === 'admin') {
    if (email.trim().toLowerCase() !== OWNER_EMAIL.toLowerCase()) {
      return res.status(403).json({ 
        success: false, 
        message: "Unauthorized: Only registered store owners can access this portal." 
      });
    }
    
    // In production, use bcrypt password validation.
    if (password !== 'admin123') {
      return res.status(401).json({ success: false, message: "Invalid Owner Password." });
    }

    const token = jwt.sign({ email, role: 'admin' }, JWT_SECRET, { expiresIn: '12h' });
    return res.json({ success: true, token, role: 'admin', ownerEmail: OWNER_EMAIL });
  } 
  
  if (role === 'customer') {
    if (!email || !password) {
      return res.status(400).json({ success: false, message: "Email and password are required." });
    }
    const token = jwt.sign({ email, role: 'customer' }, JWT_SECRET, { expiresIn: '24h' });
    return res.json({ success: true, token, role: 'customer', email });
  }

  return res.status(400).json({ success: false, message: "Invalid role specified." });
});

// Public Product Routes
app.get('/api/products', (req, res) => {
  res.json(products);
});

// Public Order Placement
app.post('/api/orders', (req, res) => {
  const { customer, items, total, paymentMethod } = req.body;
  if (!customer || !items || items.length === 0) {
    return res.status(400).json({ success: false, message: "Invalid order details." });
  }

  const newOrder = {
    id: 'ORD-' + Date.now(),
    customer,
    items,
    total,
    paymentMethod,
    status: 'Pending',
    createdAt: new Date()
  };

  orders.unshift(newOrder);

  // Return WhatsApp link payload
  let itemListStr = items.map(i => `• ${i.name} (${i.qty}kg) - ₹${i.price * i.qty}`).join('\n');
  let waMessage = `*NEW ORDER ALERT - A B Karnataka Fish Stall*\n\n` +
    `*Order ID:* ${newOrder.id}\n` +
    `*Customer Name:* ${customer.name}\n` +
    `*Phone:* ${customer.phone}\n` +
    `*Address:* ${customer.address}\n` +
    `*Payment Method:* ${paymentMethod}\n\n` +
    `*Items Ordered:*\n${itemListStr}\n\n` +
    `*Total Amount:* ₹${total}`;

  const waUrl = `https://wa.me/${OWNER_PHONE}?text=${encodeURIComponent(waMessage)}`;

  res.json({ 
    success: true, 
    orderId: newOrder.id, 
    whatsappUrl: waUrl,
    message: "Order placed successfully! Order notification queued for store owner." 
  });
});

// Protected Admin API Routes
app.get('/api/admin/orders', authenticateOwner, (req, res) => {
  res.json(orders);
});

app.post('/api/admin/products', authenticateOwner, (req, res) => {
  const { name, price, inStock, image } = req.body;
  const newProduct = {
    id: Date.now(),
    name,
    price: parseFloat(price),
    inStock: Boolean(inStock),
    image
  };
  products.push(newProduct);
  res.json({ success: true, product: newProduct });
});

app.put('/api/admin/products/:id', authenticateOwner, (req, res) => {
  const id = parseInt(req.params.id);
  const index = products.findIndex(p => p.id === id);
  if (index === -1) {
    return res.status(404).json({ success: false, message: "Product not found." });
  }

  products[index] = { ...products[index], ...req.body };
  res.json({ success: true, product: products[index] });
});

app.delete('/api/admin/products/:id', authenticateOwner, (req, res) => {
  const id = parseInt(req.params.id);
  products = products.filter(p => p.id !== id);
  res.json({ success: true, message: "Product deleted." });
});

// Serve Single Page Application
app.get('*', (req, res) => {
  res.sendFile(path.join(__dirname, 'public', 'index.html'));
});

app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
  console.log(`Owner Email configured as: ${OWNER_EMAIL}`);
});

// Login Handler
async function handleLogin(event, role) {
  event.preventDefault();
  const errorBox = document.getElementById("auth-error-msg");
  errorBox.classList.add("hidden");

  let email = "";
  let password = "";

  if (role === 'admin') {
    email = document.getElementById("admin-email").value;
    password = document.getElementById("admin-password").value;
  } else {
    email = document.getElementById("customer-email").value;
    password = document.getElementById("customer-password").value;
  }

  try {
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password, role })
    });

    const data = await res.json();

    if (!res.ok || !data.success) {
      errorBox.textContent = data.message || "Login failed.";
      errorBox.classList.remove("hidden");
      return;
    }

    // Store Auth Data
    authToken = data.token;
    userRole = data.role;
    userEmail = role === 'admin' ? data.ownerEmail : data.email;

    localStorage.setItem('fish_stall_token', authToken);
    localStorage.setItem('fish_stall_role', userRole);
    localStorage.setItem('fish_stall_email', userEmail);

    checkAuthStatus();
  } catch (err) {
    errorBox.textContent = "Network error. Server may be down.";
    errorBox.classList.remove("hidden");
  }
}

// Shopping Cart Functions
function addToCart(productId) {
  const item = productsData.find(p => p.id === productId);
  if (!item || !item.inStock) return;

  const existing = cart.find(c => c.id === productId);
  if (existing) {
    existing.qty += 1;
  } else {
    cart.push({ id: item.id, name: item.name, price: item.price, qty: 1 });
  }
  updateCartUI();
}

function updateQty(productId, change) {
  const existing = cart.find(c => c.id === productId);
  if (existing) {
    existing.qty += change;
    if (existing.qty <= 0) {
      cart = cart.filter(c => c.id !== productId);
    }
  }
  updateCartUI();
}

function updateCartUI() {
  const container = document.getElementById("cart-items-container");
  const totalDisplay = document.getElementById("cart-total-price");

  if (cart.length === 0) {
    container.innerHTML = `<p class="empty-cart-text">Your basket is empty. Select fresh fish above to start ordering!</p>`;
    totalDisplay.textContent = "₹0";
    return;
  }

  container.innerHTML = "";
  let total = 0;

  cart.forEach(item => {
    const itemTotal = item.price * item.qty;
    total += itemTotal;

    const row = document.createElement("div");
    row.className = "cart-item-row";
    row.innerHTML = `
      <div>
        <strong>${item.name}</strong>
        <br><small>₹${item.price} x ${item.qty} kg</small>
      </div>
      <div class="cart-controls">
        <button class="qty-btn" onclick="updateQty(${item.id}, -1)">-</button>
        <span>${item.qty} kg</span>
        <button class="qty-btn" onclick="updateQty(${item.id}, 1)">+</button>
      </div>
    `;
    container.appendChild(row);
  });

  totalDisplay.textContent = `₹${total}`;
  updateQRCode(total);
}

// Submit Order
async function submitOrder(event) {
  event.preventDefault();
  if (cart.length === 0) {
    alert("Please add at least one item to your cart before checking out!");
    return;
  }

  const customer = {
    name: document.getElementById("cust-name").value,
    phone: document.getElementById("cust-phone").value,
    address: document.getElementById("cust-address").value
  };

  const paymentMethod = document.getElementById("payment-method").value;
  const total = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);

  try {
    const res = await fetch('/api/orders', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ customer, items: cart, total, paymentMethod })
    });

    const data = await res.json();
    if (data.success) {
      alert("Order placed successfully! Redirecting to WhatsApp to send order summary to store owner...");
      window.open(data.whatsappUrl, '_blank');
      cart = [];
      updateCartUI();
      document.getElementById("order-form").reset();
    }
  } catch (err) {
    alert("Order submission failed. Check server connection.");
  }
}

async function updateProductOnServer(id, payload) {
  try {
    const res = await fetch(`/api/admin/products/${id}`, {
      method: 'PUT',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${authToken}`
      },
      body: JSON.stringify(payload)
    });
    if (res.ok) {
      fetchProducts();
    } else {
      const data = await res.json();
      alert(data.message);
    }
  } catch (err) {
    console.error("Update failed", err);
  }
}

  }

  return res.status(400).json({ success: false, message: "Invalid role specified." });
